/**
 * @param {string} s
 * @return {boolean}
 */
var isValid = function(s) {
    let map = {
        ")":"(",
        "]":"[",
        "}":"{"
    }
    let openers=["(","[","{"]
    let stack=[]
    let arr = s.split("")
    for (let i=0;i<arr.length;i++){
        let value = arr[i]
      
        if(openers.includes(value)){
            stack.push(value)
        }
        else  {
            if(
            map[value]==stack[stack.length-1]
            ){
                stack.pop()
            }
            else{
                return false
            }
        }
    }
    if(stack.length>0){
        return false
    }
    return true

};


//////// python ////////////////////


class Solution(object):
    def isValid(self, s):
        """
        :type s: str
        :rtype: bool
        """
        map ={
        ")":"(",
        "]":"[",
        "}":"{",
        }
        openers = ["(","[","{"]
        stack = []
        arr =list(s)
        for i in range(len(arr)):
            value = arr[i]
            if value in openers:
                stack.append(value)
            else:
                if  len(stack) and map[value]==stack[len(stack)-1]:
                    stack.pop()    
                else:
                    return False
        if (len(stack)>0):
            return False
        return True
