/**
 * @param {string[]} sentences
 * @return {number}
 */
var mostWordsFound = function(sentences) {
    let count = " "
    let x = 0
    let max=0
    for(let i=0;i<sentences.length;i++){
        for(let j=0; j<sentences[i].length; j++){
         if(sentences[i][j]==" "){
           x++
         }
        }
        if(max<x){
            max = x
        }
        x = 0
    }
    let result = max+1
    return result
};
