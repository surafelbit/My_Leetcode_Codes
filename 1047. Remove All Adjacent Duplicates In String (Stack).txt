/**
 * @param {string} s
 * @return {string}
 */
var removeDuplicates = function(s) {
    let other = s.split("")
    let array=[]
   
    for (let i =0 ; i<other.length;i++){
        if( array[array.length-1]!=other[i]){
                    array.push(other[i])
        }
        else{
            array.pop()
        }
    }
    return array.join("")
};


////////// python //////////

class Solution(object):
    def removeDuplicates(self, s):
        """
        :type s: str
        :rtype: str
        """
        other = list(s)
        array = []

        for i in range (len(other)):
            if not array or array[len(array)-1]!=other[i] :
                array.append(other[i])
            else:
                array.pop()   

        return "".join(array)
