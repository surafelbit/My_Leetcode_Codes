/**
 * @param {number[]} nums
 * @param {number} k
 * @return {number[]}
 */
var topKFrequent = function(nums, k) {
    let map={}
   for (let i=0; i<nums.length ; i++){
    let current = nums[i];
    map[current] = (map[current] || 0) + 1;
   } 
   
   let arr=[]
   while (k > 0) {
    const keys = Object.keys(map);
    if (keys.length === 0) break;

    let high = keys[0]; 

    // Find the ACTUAL highest in this current map
    for (let key in map) {
      if (map[key] > map[high]) {
        high = key;
      }
    }

    // 3. ONLY once the for-loop is DONE, we take the winner
    arr.push(Number(high));
    delete map[high]; // Remove the winner so the next while-loop finds the runner-up
    
    k--;
  }
  return arr
}
