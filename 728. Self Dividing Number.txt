class Solution(object):
    def removeDuplicates(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        if(len(nums)<=2):
            return len(nums)
        reader = 2
        writer = 2    
        while(reader<len(nums)):
            if(nums[writer-2]==nums[reader]):
                reader= reader+1
            else:
                nums[writer]=nums[reader]
                reader=reader+1
                writer=writer+1
        return writer
