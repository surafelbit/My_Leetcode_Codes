/**
 * @param {string} s
 * @param {string} t
 * @return {boolean}
 */
var backspaceCompare = function(s, t) {
    let array = []
    let array2 = []
    for (let i=0; i<s.length;i++){
        if(s[i]!="#"){
            array.push(s[i])
        }else{
            array.pop()
        }
    }
     for (let i=0; i<t.length;i++){
        if(t[i]!="#"){
            array2.push(t[i])
        }else{
            array2.pop()
        }
    }
    if(array.length!=array2.length){
        return false
    }
    for(let i=0;i<array.length;i++){
        if(array[i]!=array2[i]){
            return false
        }
    }
    return true
};
