var addBinary = function(a, b) {
    var s = a.length
    var k = b.length
    
    var result =""

b = "0".repeat(s - k) + b      
    }
    else {
a = "0".repeat(k - s) + a
   
    }
    let carry 
    for (let i =a.length-1 ;i>=0; i--){

    if(a[i]=="1" &&b[i]=="1"){
        if(carry){
            result = "1"+result
        }
        else{
            result = "0"+result
        
        }
        carry =1
    }
      else if(a[i]!=b[i]){
        if(carry){
                        result = "0"+result

                 carry =1

       }

        }

    }

        }

 carry=0
    }
    }
    if(carry){

    }

};
