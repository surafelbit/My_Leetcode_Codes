/**
 * @param {number[]} nums1
 * @param {number[]} nums2
 * @return {number[]}
 */
var intersect = function(nums1, nums2) {
    let map={}
    let mapA={}
    let result = []
    for(let i =0; i<nums1.length; i++){
    if(!map[nums1[i]]){
        map[nums1[i]]=1
    }
    else{
        map[nums1[i]]++
    }
    }
    for(let i = 0; i<nums2.length; i++){
      if(!mapA[nums2[i]]){
        mapA[nums2[i]]=1
    }
    else{
        mapA[nums2[i]]++
    }  
    }
    for (let x in mapA){
    if(mapA[x] && map[x]){
         for (let y = 0 ; y<Math.min(mapA[x],map[x]); y++){
          result.push(Number(x))
          }
    }
    }
    
    return result
};
