/**
 * @param {number[]} nums
 * @return {number[]}
 */
var findDuplicates = function(nums) {
    let map={}
    let result=[]
    for (let i=0;i<nums.length;i++){
         if(!map[nums[i]]){
            map[nums[i]]=true
         }
         else{
            result.push(nums[i])
         }
    }
    return result
};
