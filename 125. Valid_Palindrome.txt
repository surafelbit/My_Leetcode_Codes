var isPalindrome = function(s) {
    var x = s.toLowerCase()
    var cleaned = x.replace(/[^a-zA-Z0-9]/g, "");
    var r = Math.floor(cleaned.length / 2);
    for ( let i=0; i<=r; i++){
        if (cleaned[i]!= cleaned[cleaned.length-i-1] )

    }
    return true
};

/**
 * @param {string} s
 * @return {boolean}
 */
var isPalindrome = function(s) {
    
    var x = s.toLowerCase()
     var cleaned = x.replace(/[^a-zA-Z0-9]/g, "");
     let left = 0
    let right = cleaned.length-1
    while(right>left){
        if(cleaned[right]!=cleaned[left]){
            return false
        }
        left++
        right--
    }
    return true
};
