/**
 * @param {string} s
 * @return {string}
 */
var reverseVowels = function(s) {
  let holder = ['a','e','i','o','u','A','E','I','O','U']

  let left = 0
  let right = s.length - 1

  let arr = s.split('') // needed because strings can't be swapped

  while(left < right){

    if(!holder.includes(arr[left]) && !holder.includes(arr[right])){
      left++
      right--
    }
    else if(holder.includes(arr[left]) && !holder.includes(arr[right])){
      right--
    }
    else if(!holder.includes(arr[left]) && holder.includes(arr[right])){
      left++
    }
    else{
      [arr[left], arr[right]] = [arr[right], arr[left]]
      left++
      right--
    }

  }

  return arr.join('')
};
