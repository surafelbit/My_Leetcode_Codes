/**
 * @param {number[]} nums
 * @param {number} k
 * @return {number}
 */
var findMaxAverage = function(nums, k) {
    let sum= 0
    for( let i=0;i<k;i++){
        sum +=nums[i]
    }
    let maxsum = sum
    let i = 1
    while(i+k-1<=nums.length){
    sum = sum + nums[i+k-1] - nums[i-1]
       if(sum>maxsum){
        maxsum = sum
      }
      i++
    }

    return maxsum/k
};
