/**
 * @param {number[]} height
 * @return {number}
 */
var maxArea = function(height) {
    let area = 0
    let biggestArea = 0
    let left = 0
    let right = height.length-1
    while(left<right){
        area= (right-left)*Math.min(height[right],height[left])
        if(area>biggestArea){
            biggestArea=area
        }

    if(height[left]>height[right]){
        right--
    }
    else{
        left++
    }
}
return biggestArea

};

class Solution(object):
    def maxArea(self, height):
        """
        :type height: List[int]
        :rtype: int
        """
        left = 0
        right = len(height)-1
        biggestArea=0
        while(right>left):
            area=(right-left)*(min(height[left],height[right]))
            if(area>biggestArea):
                biggestArea=area
            if(height[right]>height[left]):
                left = left+1
            else:
                right = right - 1 
        return biggestArea
