/**
 * @param {number[]} nums
 * @param {number} target
 * @return {number}
 */
var searchInsert = function(nums, target) {
    let left = 0
    let right = nums.length - 1
    let mid = 0
    while(left<=right){
        mid = left + Math.floor((right-left)/2)
        if(nums[mid]>target){
            right = mid-1
        }
        else if(nums[mid]<target){
            left = mid+1
        }
        else if(nums[mid]==target){
            return mid
        }
       
    }
    return left
};


//////////// python ////////////////

class Solution(object):
    def searchInsert(self, nums, target):
        """
        :type nums: List[int]
        :type target: int
        :rtype: int
        """
        left = 0
        right = len(nums)-1
        mid = 0
        while(left<=right):
            mid = left + right-left//2
            if(nums[mid]>target):
                right = mid-1
            elif(nums[mid]<target):
                left = mid+1
            elif(nums[mid]==target):
                return mid    
        return left
