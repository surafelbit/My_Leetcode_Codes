var nextGreaterElement = function(nums1, nums2) {
    let arr = []
    let map={}
        let stack = []

    for (let t=0; t<nums2.length; t++){
    let number = nums2[t]
    while(stack.length && number>stack[stack.length-1]){
     let smaller = stack.pop();
     map[smaller]= number
    }
    stack.push(number)
    }
    for (let i =0; i<nums1.length; i++){
        if(map[nums1[i]]){
            arr.push(map[nums1[i]])
        }
        else{
            arr.push(-1)
        }
    }
    
    return arr
};

class Solution(object):
    def nextGreaterElement(self, nums1, nums2):
        """
        :type nums1: List[int]
        :type nums2: List[int]
        :rtype: List[int]
        """
        stack = []
        arr = []
        map = {}
        for i in range(len(nums2)):
            number = nums2[i]
            while(len(stack )and number > stack[(len(stack)-1)] ):
                smaller = stack.pop()
                map[smaller] = number
            stack.append(number)    
        for i in range(len(nums1)):
            if(nums1[i] in map):
                arr.append(map[nums1[i]])
            else:
                arr.append(-1)
        return arr
