class Solution(object):
    def containsDuplicate(self, nums):
        """
        :type nums: List[int]
        :rtype: bool
        """
        x = set()
        for i in range (len(nums)):
         
            if nums[i] in x:
                return True
            x.add(nums[i])
        return False
