class Solution(object):
    def isPalindrome(self, x):
        """
        :type x: int
        :rtype: bool
        """
        y = str(x)
        v = len(y)-1
        isPalindrome = True
        for i in range(v):
            if(y[i]!=y[v]):
                isPalindrome =False
                
                
            v= v-1
        return isPalindrome
