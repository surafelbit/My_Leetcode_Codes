/**
 * @param {number[]} nums
 * @param {number} k
 * @return {boolean}
 */
var containsNearbyDuplicate = function(nums, k) {
    let map={}
    for (let i = 0; i<nums.length; i++){
    if(map[nums[i]]==undefined){
        map[nums[i]]=i
    }
    else{
      if(Math.abs(map[nums[i]]-i)<=k){
        return true
      }
      else{
        map[nums[i]]=i
      }
    }

    }
          return false

};


////////////////// python /////////////////////
class Solution(object):
    def containsNearbyDuplicate(self, nums, k):
        """
        :type nums: List[int]
        :type k: int
        :rtype: bool
        """
        map={}
        for i in range (len(nums)):
            if nums[i] not in map:
                map[nums[i]]=i
            
            else:
                if(abs(map[nums[i]]-i)<=k):
                    return True
                else:
                    map[nums[i]]=i    
        return False
