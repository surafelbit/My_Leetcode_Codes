/**
 * @param {string} s
 * @return {number}
 */
var lengthOfLastWord = function(s) {
    var x = s.split(" ")
    var y = x.filter(Boolean)
    
    return y[y.length-1].length
    
};

///// python//////////////


class Solution(object):
    def lengthOfLastWord(self, s):
        """
        :type s: str
        :rtype: int
        """
        x = s.split(" ")
        y = list(filter(bool,x))
        return len(y[len(y)-1])
